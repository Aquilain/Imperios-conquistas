LÓGICA DE RECONQUISTA CUANDO ROMA GANA LA GUERRA // =================================================== /** * Función que se ejecuta cuando Roma gana una guerra/campaña decisiva. * Recupera automáticamente los territorios históricos perdidos (Magna Grecia, Sicilia, etc.) */ function alGanarGuerraRoma() { console.log("¡Roma ha ganado la guerra! Iniciando proceso de reconquista territorial..."); // Lista de territorios reclamados históricamente por Roma al ganar const territoriosARecuperar = ['magna_grecia', 'sicilia', 'corcega']; territoriosARecuperar.forEach((idTerritorio) => { if (territorios[idTerritorio]) { // Cambiar control a Roma territorios[idTerritorio].faccion = 'roma'; // Asignar una guarnición romana mínima de reconquista si tenía pocas tropas territorios[idTerritorio].tropas = Math.max(territorios[idTerritorio].tropas, 25); console.log(`- Roma ha recuperado: ${territorios[idTerritorio].nombre}`); } }); // Redibujar el mapa para reflejar los cambios renderizarMapa(); } /** * Función para resolver un combate/guerra entre facciones. */ function resolverGuerra(atacanteId, defensorId) { const atacante = territorios[atacanteId]; const defensor = territorios[defensorId]; if (!atacante || !defensor) return; // Lógica de resolución de batalla simple if (atacante.tropas > defensor.tropas) { const faccionAnterior = defensor.faccion; defensor.faccion = atacante.faccion; defensor.tropas = Math.floor(atacante.tropas - defensor.tropas * 0.5); atacante.tropas = Math.floor(atacante.tropas * 0.3); console.log(`¡${atacante.nombre} ha conquistado ${defensor.nombre}!`); // Si el atacante es Roma y ha derrotado una posición clave de Cártago if (atacante.faccion === 'roma' && faccionAnterior === 'cartago') { alGanarGuerraRoma(); } else { renderizarMapa(); } } else { defensor.tropas -= Math.floor(atacante.tropas * 0.5); atacante.tropas = 5; console.log(`La defensa de ${defensor.nombre} ha resistido el ataque.`); renderizarMapa(); } } // =================================================== // 4. RENDERIZADO EN CANVAS (DIBUJO DEL MAPA) // =================================================== function renderizarMapa() { const canvas = document.getElementById('mapaCanvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); // Dibujar cada territorio Object.keys(territorios).forEach((id) => { const region = territorios[id]; const colorFacction = FACCIONES[region.faccion] ? FACCIONES[region.faccion].color : '#cccccc'; // Dibujar polígono del territorio ctx.beginPath(); region.poligono.forEach((pt, index) => { if (index === 0) ctx.moveTo(pt.x, pt.y); else ctx.lineTo(pt.x, pt.y); }); ctx.closePath(); // Relleno ctx.fillStyle = colorFacction; ctx.fill(); // Borde ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 2; ctx.stroke(); // Calcular centro para el texto let centroX = 0, centroY = 0; region.poligono.forEach(pt => { centroX += pt.x; centroY += pt.y; }); centroX /= region.poligono.length; centroY /= region.poligono.length; // Etiqueta Nombre ctx.fillStyle = '#ffffff'; ctx.font = 'bold 12px Arial'; ctx.textAlign = 'center'; ctx.fillText(region.nombre.toUpperCase(), centroX, centroY - 5); // Etiqueta Tropas (Espadas) ctx.font = '11px Arial'; ctx.fillText(`⚔️ ${region.tropas}`, centroX, centroY + 12); }); }
