// --- ESTADO DEL JUEGO Y CONFIGURACIÓN BASE --- let currentYear = -218; // 218 a.C. let playerFaction = "Roma"; const gameState = { roma: { owner: "Roma", troops: 10, fortress: 2 }, cartago: { owner: "Cartago", troops: 12, fortress: 1 }, sagunto: { owner: "Rebeldes", troops: 3, fortress: 1 } }; const REGION_DEFINITIONS = { roma: { name: "Roma", adj: ["sagunto"], baseIncome: 15 }, cartago: { name: "Cartago", adj: ["sagunto"], baseIncome: 12 }, sagunto: { name: "Sagunto", adj: ["roma", "cartago"], baseIncome: 8 } }; // --- FUNCIONES AUXILIARES DE DIPLOMACIA Y SISTEMA --- function isAtWar(factionA, factionB) { if (factionA === factionB) return false; // Lógica simplificada de diplomacia return (factionA === "Cartago" && factionB === "Roma") || (factionA === "Roma" && factionB === "Cartago") || factionB === "Rebeldes"; } function isFactionInAnyWar(faction) { return true; // Estado estático de prueba } function getRegionDisplayName(key) { return REGION_DEFINITIONS[key]?.name || key; } function formatYear(year) { return year < 0 ? `${Math.abs(year)} a.C.` : `${year} d.C.`; } function playSound(soundKey) { // Implementación del reproductor de audio console.log(`[Audio]: Reproduciendo ${soundKey}`); } // --- LOGICA DEL TURNO DE LA IA --- function processAITurn() { Object.keys(gameState).forEach(key => { const reg = gameState[key]; const def = REGION_DEFINITIONS[key]; if (reg.owner !== playerFaction && reg.troops > 2) { const targetKey = def.adj.find(adjKey => { const neighbor = gameState[adjKey]; // REALISMO HISTÓRICO: Cartago no puede conquistar la capital Roma de forma libre por IA if (reg.owner === "Cartago" && adjKey === "roma") { return false; } if (neighbor.owner !== reg.owner && isAtWar(reg.owner, neighbor.owner)) { const aiAttackPower = reg.troops - 1; const enemyDefensePower = neighbor.troops + (neighbor.fortress * 2); return aiAttackPower > enemyDefensePower; } return false; }); if (targetKey) { const target = gameState[targetKey]; const aiAttackingTroops = reg.troops - 1; const enemyDefensePower = target.troops + (target.fortress * 2); const casualties = Math.floor(enemyDefensePower * 0.5); const defenderName = target.owner; target.owner = reg.owner; target.troops = Math.max(1, aiAttackingTroops - casualties); target.fortress = Math.max(0, target.fortress - 1); reg.troops = 1; if (defenderName === playerFaction) { playSound('battle_lose'); logMessage(`🚨 ¡INVASIÓN ENEMIGA! ${reg.owner} capturó ${getRegionDisplayName(targetKey)}.`, 'ai'); } else { logMessage(`⚔️ HISTORIA: ${reg.owner} arrebató ${getRegionDisplayName(targetKey)} a ${defenderName}.`, 'ai'); } } } }); } // --- INTERFAZ Y TOOLTIPS --- function showTooltip(e, key) { const tooltip = document.getElementById('tooltip'); const reg = gameState[key]; const def = REGION_DEFINITIONS[key]; if (!reg || !def) return; const atWarWithPlayer = isAtWar(playerFaction, reg.owner); const isOwnerAtWar = isFactionInAnyWar(reg.owner); let warStatusTag = ""; if (reg.owner === playerFaction) { warStatusTag = `(Tu Región)`; } else if (atWarWithPlayer) { warStatusTag = `⚔️ (En Guerra Contigo)`; } else if (isOwnerAtWar) { warStatusTag = `⚔️ (En Guerra)`; } else { warStatusTag = `🕊 (En Paz)`; } const baseInc = def.baseIncome || 10; const totalInc = baseInc + (reg.fortress * 5); tooltip.innerHTML = ` ${getRegionDisplayName(key)} ${warStatusTag}
Facción: ${reg.owner}
Guarnición: ${reg.troops} | Fortaleza: Lvl ${reg.fortress}
Riqueza Regional: +${totalInc} 💰/año
Poder Defensivo: ${reg.troops + (reg.fortress * 2)} `; tooltip.style.display = 'block'; tooltip.style.left = (e.pageX + 15) + 'px'; tooltip.style.top = (e.pageY + 15) + 'px'; } function hideTooltip() { const tooltip = document.getElementById('tooltip'); if (tooltip) tooltip.style.display = 'none'; } function logMessage(msg, type = 'system') { const box = document.getElementById('game-log'); if (!box) return; const entry = document.createElement('div'); entry.className = `log-entry ${type}`; entry.textContent = `[${formatYear(currentYear)}] ${msg}`; box.appendChild(entry); box.scrollTop = box.scrollHeight; }
